function doGet() { // Lấy sheet đầu tiên hoặc sheet chứa chữ "Job" var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Job Board") || ss.getSheets()[0]; var data = sheet.getDataRange().getValues(); var jobs = []; // Tự động quét từ hàng 3 trở đi for (var i = 2; i < data.length; i++) { var row = data[i]; // Đọc các giá trị cột var code = row[0] ? row[0].toString().trim() : ""; // Cột A: Mã var title = row[4] ? row[4].toString().trim() : ""; // Cột E: Vị trí var linkJD = row[6] ? row[6].toString().trim() : "#"; // Cột G: Link JD var city = row[7] ? row[7].toString().trim() : ""; // Cột H: Thành phố var district = row[8] ? row[8].toString().trim() : ""; // Cột I: Địa điểm var salary = row[9] ? row[9].toString().trim() : ""; // Cột J: Mức lương var status = row[10] ? row[10].toString().trim() : ""; // Cột K: Hiện trạng var category = row[11] ? row[11].toString().trim() : ""; // Cột L var type = row[12] ? row[12].toString().trim() : ""; // Cột M var hot = row[13] ? row[13].toString().trim() : ""; // Cột N var workTime = row[14] ? row[14].toString().trim() : ""; // Cột O var description = row[15] ? row[15].toString().trim() : ""; // Cột P var requirements = row[16] ? row[16].toString() : ""; // Cột Q var benefits = row[17] ? row[17].toString() : ""; // Cột R // Lấy công việc nếu có Tên vị trí VÀ (Trạng thái là Đang tuyển HOẶC không bị Closed) var isAvailable = status.toLowerCase().indexOf("đang tuyển") !== -1 || (status !== "" && status.toLowerCase().indexOf("closed") === -1); if (title !== "" && isAvailable) { jobs.push({ id: code || ("job-" + i), title: title, category: category || "Chung", location: (city || "") + (district ? " (" + district + ")" : ""), type: type || "Full-time", salary: salary || "Thỏa thuận", linkJD: linkJD || "#", hot: (hot.toLowerCase() === "true" || hot.toLowerCase() === "x"), workTime: workTime, description: description || "Bấm để xem file JD chi tiết.", requirements: requirements ? requirements.split(";") : [], benefits: benefits ? benefits.split(";") : [] }); } } return ContentService.createTextOutput(JSON.stringify(jobs)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var contents = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("CV tự nguyện") || ss.getSheets()[0]; var timestamp = new Date(); sheet.appendRow([ timestamp, contents.name || "", contents.contact || "", contents.jobTitle || contents.service || "", contents.cvLink || contents.note || "", "Nguồn từ Website Bà Mai" ]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
